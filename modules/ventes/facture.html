/*==================================================== IMPRIMER FACTURE ====================================================*/ function imprimerFacture(id) { const vente = trouverVente(id); if (!vente) { alert( "Vente introuvable." ); return; } const fenetre = window.open( "", "_blank" ); if (!fenetre) { alert( "Impossible d'ouvrir la fenêtre d'impression." ); return; } fenetre.document.write(` Facture

FERME ASHER

FACTURE DE VENTE

Facture : ${vente.facture || vente.id}

Date : ${vente.date}

Client : ${vente.client}

Téléphone : ${vente.telephone || "-"}

Produit ${vente.produit}
Quantité ${vente.quantite}
Prix unitaire ${formatFC(vente.prix)}
Remise ${formatFC(vente.remise)}
Paiement ${vente.paiement}

TOTAL : ${formatFC(vente.total)}


Merci pour votre confiance.

`); fenetre.document.close(); fenetre.onload = function() { fenetre.print(); }; }
